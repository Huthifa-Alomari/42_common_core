===== ft_split.c =====
int main(void)
{
	char	**arr;
	int		i;

	arr = ft_split("hello world foo", ' ');
	i = 0;
	while (arr[i])
	{
		printf("%s\n", arr[i]);
		free(arr[i]);
		i++;
	}
	free(arr);
	return (0);
}

===== ft_strmapi.c =====
char	to_upper(unsigned int i, char c)
{
	(void)i;
	if (c >= 'a' && c <= 'z')
		return (c - 32);
	return (c);
}

int main(void)
{
	char	*s;

	s = ft_strmapi("hello", to_upper);
	printf("%s\n", s);
	free(s);
	return (0);
}

===== ft_lstmap.c =====
void	*upper(void *content)
{
	return (content);
}

void	del(void *content)
{
	(void)content;
}

int main(void)
{
	t_list	*lst;
	t_list	*new;

	lst = ft_lstnew("a");
	ft_lstadd_back(&lst, ft_lstnew("b"));
	new = ft_lstmap(lst, upper, del);
	printf("%s\n", (char *)new->content);
	return (0);
}
